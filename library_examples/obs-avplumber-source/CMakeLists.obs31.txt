
cmake_minimum_required(VERSION 3.16...3.25)

project(avplumber-source VERSION 1.0)

include_directories(SYSTEM "${CMAKE_SOURCE_DIR}/libobs")

set(CMAKE_INCLUDE_CURRENT_DIR ON)

add_library(avplumber-source MODULE)

target_sources(avplumber-source PRIVATE avplumber-source.cpp)


set(EMBED_IN obs CACHE INTERNAL "")
set(AVPLUMBER_BUILD_EXE OFF CACHE INTERNAL "")
add_subdirectory(avplumber)


target_link_libraries(avplumber-source
	${avplumber-source_PLATFORM_DEPS}
	avplumber_static
	libobs)

set_target_properties_obs(avplumber-source PROPERTIES FOLDER plugins/avplumber-source PREFIX "")
